```ts
/**
 * "Public source, N tests" on the landing page (wiring-spec W4.3: no figure without a live source): N is counted from the
 * contract's Foundry test files at build time (functions named test* or invariant*). null if they aren't there, and the
 * page then leaves the number out.
 */
function contractTestCount(): number | null {
  const dir = fileURLToPath(new URL("../test", import.meta.url));
  try {
    const n = fs
      .readdirSync(dir)
      .filter((f) => f.endsWith(".t.sol"))
      .reduce((sum, f) => sum + (fs.readFileSync(path.join(dir, f), "utf8").match(/^\s*function (test|invariant)\w*\s*\(/gm)?.length ?? 0), 0);
    return n > 0 ? n : null;
  } catch {
    return null;
  }
}

export default defineConfig({
  plugins: [react()],
  define: { __CONTRACT_TEST_COUNT__: JSON.stringify(contractTestCount()) },
  server: { port: 5173, strictPort: true },
```

### `scripts/secret-scan.mjs`

```js
// Secret scan (wiring-spec section 1): fails if a backend secret could reach the browser.
//
//   node scripts/secret-scan.mjs          scan src/, index.html, public/, .env.example, the env files' keys, dist/ if built
//
// Three checks, all printing names and file:line only, never a value:
//  1. Forbidden names: the service-role key, the verifier key, the X bearer token or any *private key* name, in source,
//     in the built bundle, or as a key in any frontend env file.
//  2. 64-hex strings (a private key is one): every 0x + 64 hex (or bare 64 hex) in src/ and dist/ must be listed in
//     scripts/secret-scan-allow.txt with the reason it is public (tx hashes, id hashes, library curve constants).
//  3. Env values: no VITE_* value in the env files for development or production may look like a private key, since
//     Vite would put it in the bundle.
import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import { loadEnv } from "vite";

const ROOT = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");
// Built from parts so this file doesn't trip its own check.
const FORBIDDEN = new RegExp(["SERVICE" + "_ROLE", "VERIFIER" + "_PRIVATE", "X_BEARER" + "_TOKEN", "PRIVATE" + "_KEY", "DEPLOYER" + "_KEY"].join("|"), "i");
const HEX64 = /(?<![0-9a-fA-Fx])(?:0x)?([0-9a-fA-F]{64})(?![0-9a-fA-F])/g;

const allow = new Set(
  fs
    .readFileSync(path.join(ROOT, "scripts/secret-scan-allow.txt"), "utf8")
    .split("\n")
    .map((l) => l.trim().split(/\s+/)[0])
    .filter((v) => /^0x[0-9a-f]{64}$/i.test(v ?? ""))
    .map((v) => v.slice(2).toLowerCase()),
);

function* files(dir) {
  if (!fs.existsSync(dir)) return;
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    const p = path.join(dir, e.name);
    if (e.isDirectory()) yield* files(p);
    else if (/\.(tsx?|jsx?|mjs|cjs|html|css|json|txt|svg|map|example)$/.test(e.name) || e.name.startsWith(".env.example")) yield p;
  }
}

const problems = [];
const rel = (p) => path.relative(ROOT, p).replace(/\\/g, "/");

const targets = [
  ...files(path.join(ROOT, "src")),
  ...files(path.join(ROOT, "public")),
  ...files(path.join(ROOT, "dist")),
  path.join(ROOT, "index.html"),
  path.join(ROOT, ".env.example"),
].filter((p) => fs.existsSync(p));

for (const file of targets) {
  const lines = fs.readFileSync(file, "utf8").split("\n");
  lines.forEach((line, i) => {
    const where = `${rel(file)}:${i + 1}`;
    const name = line.match(FORBIDDEN);
    if (name) problems.push(`${where}: forbidden name ${name[0]}`);
    for (const m of line.matchAll(HEX64)) {
      if (!allow.has(m[1].toLowerCase())) problems.push(`${where}: 64-hex string ${m[1].slice(0, 6)}… is not in scripts/secret-scan-allow.txt`);
    }
  });
}

// Env files: keys for every mode (loadEnv with an empty prefix returns all keys), values for VITE_* only.
for (const mode of ["development", "production"]) {
  const env = loadEnv(mode, ROOT, "");
  const fromFiles = Object.keys(env).filter((k) => !(k in process.env) || k.startsWith("VITE_"));
  for (const k of fromFiles) {
    if (FORBIDDEN.test(k)) problems.push(`env (${mode}): forbidden variable name ${k} in a frontend env file`);
    if (k.startsWith("VITE_") && /^(0x)?[0-9a-fA-F]{64}$/.test(String(env[k]).trim())) problems.push(`env (${mode}): ${k} looks like a private key`);
  }
}

if (problems.length) {
  console.error(`secret-scan: ${problems.length} problem(s)\n  ${problems.join("\n  ")}`);
  process.exit(1);
}
console.log(`secret-scan: ok (${targets.length} files${fs.existsSync(path.join(ROOT, "dist")) ? ", dist included" : ", no dist yet"})`);
```

### `src/env.ts`

```ts
// Frontend environment (docs/design/wiring-spec.md section 1), validated once at startup.
//
// Every value here is public: Vite only exposes VITE_* variables, and they end up in the bundle. Supabase's service key,
// the claim verifier's key, the X API token and any wallet key must never be set here (scripts/secret-scan.mjs fails
// the build if one is).
//
// The chain values default to the live deployment (deployment.json; src/chain/config.test.ts fails if they drift), so a
// build without a .env.local still talks to the right contract. A value that is set but malformed is an error, shown on
// one screen instead of a blank page. The Supabase values are optional: without them the app says the backend isn't
// connected.
import { getAddress, isAddress } from "viem";

const raw = import.meta.env;

export interface Env {
  chainId: number;
  rpcUrl: string;
  explorerUrl: string;
  escrowAddress: `0x${string}`;
  escrowDeployBlock: bigint;
  supabaseUrl: string | null;
  supabaseAnonKey: string | null;
  functionsUrl: string | null;
  reownProjectId: string | null;
  appOrigin: string | null;
}

const DEFAULTS = {
  VITE_CHAIN_ID: "6281971",
  VITE_RPC_URL: "https://rpc.testnet.dogeos.com/",
  VITE_EXPLORER_URL: "https://blockscout.testnet.dogeos.com",
  VITE_ESCROW_ADDRESS: "0x207652b0767F9f19d24B5300E2123D415167aCeF",
  VITE_ESCROW_DEPLOY_BLOCK: "9065839",
} as const;

const trimSlash = (s: string) => s.replace(/\/+$/, "");
const isHttpUrl = (s: string) => {
  try {
    const u = new URL(s);
    return u.protocol === "https:" || u.protocol === "http:";
  } catch {
    return false;
  }
};

/** Pure, so it can be unit-tested with any input. Returns the parsed env or the list of problems (names only, never values). */
export function parseEnv(src: Record<string, string | boolean | undefined>): { env: Env; errors: [] } | { env: null; errors: string[] } {
  // Each value is read as VITE_X, or as VITE_REACT_APP_X when VITE_X is unset: Noah AI prefixes every env variable
  // with VITE_REACT_APP_ (docs.trynoah.ai/integrations/environment-variables). Vercel and .env.local keep using VITE_X.
  const one = (k: string) => {
    const v = src[k];
    return typeof v === "string" && v.trim() ? v.trim() : null;
  };
  const get = (k: string) => one(k) ?? one(k.replace(/^VITE_/, "VITE_REACT_APP_"));
  const errors: string[] = [];
  const req = (k: keyof typeof DEFAULTS) => get(k) ?? DEFAULTS[k];

  const chainId = Number(req("VITE_CHAIN_ID"));
  if (!Number.isSafeInteger(chainId) || chainId <= 0) errors.push("VITE_CHAIN_ID must be a positive integer");
  const rpcUrl = req("VITE_RPC_URL");
  if (!isHttpUrl(rpcUrl)) errors.push("VITE_RPC_URL must be an http(s) URL");
  const explorerUrl = trimSlash(req("VITE_EXPLORER_URL"));
  if (!isHttpUrl(explorerUrl)) errors.push("VITE_EXPLORER_URL must be an http(s) URL");
  const escrow = req("VITE_ESCROW_ADDRESS");
  if (!isAddress(escrow, { strict: false })) errors.push("VITE_ESCROW_ADDRESS must be a 0x address");
  const block = req("VITE_ESCROW_DEPLOY_BLOCK");
  if (!/^\d{1,12}$/.test(block)) errors.push("VITE_ESCROW_DEPLOY_BLOCK must be a block number");

  const supabaseUrl = get("VITE_SUPABASE_URL");
  if (supabaseUrl && !isHttpUrl(supabaseUrl)) errors.push("VITE_SUPABASE_URL must be an http(s) URL");
  const supabaseAnonKey = get("VITE_SUPABASE_ANON_KEY");
  if (supabaseUrl && !supabaseAnonKey) errors.push("VITE_SUPABASE_ANON_KEY is required when VITE_SUPABASE_URL is set");
  const functionsUrl = get("VITE_FUNCTIONS_URL") ?? (supabaseUrl ? `${trimSlash(supabaseUrl)}/functions/v1` : null);
  if (functionsUrl && !isHttpUrl(functionsUrl)) errors.push("VITE_FUNCTIONS_URL must be an http(s) URL");
  const appOrigin = get("VITE_APP_ORIGIN");
  if (appOrigin && !isHttpUrl(appOrigin)) errors.push("VITE_APP_ORIGIN must be an http(s) origin, e.g. https://muchtip.vercel.app");

  if (errors.length) return { env: null, errors };
  return {
    env: {
      chainId,
      rpcUrl,
      explorerUrl,
      escrowAddress: getAddress(escrow),
      escrowDeployBlock: BigInt(block),
      supabaseUrl: supabaseUrl ? trimSlash(supabaseUrl) : null,
      supabaseAnonKey,
      functionsUrl: functionsUrl ? trimSlash(functionsUrl) : null,
      reownProjectId: get("VITE_REOWN_PROJECT_ID"),
      appOrigin: appOrigin ? trimSlash(appOrigin) : null,
    },
    errors: [],
  };
}

const parsed = parseEnv(raw as unknown as Record<string, string | undefined>);
/** Problems found at startup; main.tsx shows them instead of the app. */
export const ENV_ERRORS: string[] = parsed.errors;
/** The validated env. When ENV_ERRORS is non-empty this holds the defaults only so modules can load; the app never renders. */
export const ENV: Env = parsed.env ?? (parseEnv({}).env as Env);
```

```ts
import { BaseError, ContractFunctionRevertedError, InsufficientFundsError, UserRejectedRequestError } from "viem";

export type TxErrorKind = "rejected" | "revert" | "funds" | "rpc" | "unknown";
export interface TxError {
  kind: TxErrorKind;
  /** The contract's custom error name for a revert, e.g. "BelowMinTip". */
  revert?: string;
  title: string;
  detail: string;
}

/** Values the copy fills in: the chain's minimum ("5 DOGE") and, for TooEarly, when the refund opens ("14:32"). */
export interface ErrorContext {
  min?: string;
  when?: string;
}

// Wiring-spec section 6: the exact copy for every TipEscrow custom error (src/TipEscrow.sol).
const REVERTS: Record<string, (ctx: ErrorContext) => string> = {
  BelowMinTip: ({ min }) => (min ? `The minimum tip is ${min}.` : "This amount is below the contract's minimum tip."),
  ZeroValue: () => "Enter an amount.",
  NothingToClaim: () => "There's nothing to claim right now.",
  Expired: () => "That claim took too long. Try again.",
  NonceUsed: () => "This claim can't be used. Try again.",
  BadSignature: () => "This claim can't be used. Try again.",
  TooEarly: ({ when }) => (when ? `You can refund this tip at ${when}.` : "This tip can't be refunded yet."),
  AlreadyClaimed: () => "They already claimed this tip.",
  AlreadyRefunded: () => "You already got this tip back.",
  NotTipper: () => "Only the person who sent this tip can refund it.",
  UnknownTip: () => "Only the person who sent this tip can refund it.",
  ZeroAddress: () => "The wallet address is missing.",
  TransferFailed: () => "The contract couldn't send DOGE to that wallet.",
};

export const CANCELLED = "You cancelled in your wallet.";
export const NOT_ENOUGH = "Not enough DOGE to cover the tip and the network fee.";
export const OFFLINE = "Can't reach the server. Check your connection.";

/** The copy for a contract revert by its custom error name, or null for an unknown name. */
export const revertMessage = (name: string, ctx: ErrorContext = {}) => REVERTS[name]?.(ctx) ?? null;

/** Turns any wagmi/viem error into what the UI shows. Never leaks raw RPC text as the headline. */
export function describeTxError(err: unknown, ctx: ErrorContext = {}): TxError {
  if (err instanceof BaseError) {
    if (err.walk((e) => e instanceof UserRejectedRequestError) || /rejected|denied/i.test(err.shortMessage) || (err.walk() as { code?: number }).code === 4001)
      return { kind: "rejected", title: CANCELLED, detail: "Nothing was sent." };
    const revert = err.walk((e) => e instanceof ContractFunctionRevertedError);
    if (revert instanceof ContractFunctionRevertedError) {
      const name = revert.data?.errorName;
      if (name) return { kind: "revert", revert: name, title: "The contract refused it", detail: revertMessage(name, ctx) ?? "The contract refused the transaction." };
      return { kind: "revert", title: "The contract refused it", detail: revert.shortMessage };
    }
    if (err.walk((e) => e instanceof InsufficientFundsError) || /insufficient funds/i.test(err.message))
      return { kind: "funds", title: "Not enough DOGE", detail: NOT_ENOUGH };
    if (/fetch|network|timeout|HTTP request failed|could not be found|rpc/i.test(err.message))
      return { kind: "rpc", title: "The network didn't respond", detail: OFFLINE };
    return { kind: "unknown", title: "Something went wrong", detail: err.shortMessage };
  }
  if ((err as { code?: number } | null)?.code === 4001) return { kind: "rejected", title: CANCELLED, detail: "Nothing was sent." };
  return { kind: "unknown", title: "Something went wrong", detail: err instanceof Error ? err.message : String(err) };
}
```

### `src/chain/fees.ts`

```ts
import { encodeFunctionData, parseEther, type Address, type Hex, type PublicClient } from "viem";
import { generatePrivateKey, privateKeyToAccount } from "viem/accounts";
import { escrowAbi, oracleAbi } from "./abi";
import { CHAIN_ID, ESCROW_ADDRESS, L1_GAS_ORACLE } from "./config";
import { QUOTE_SAMPLE } from "../data/samples";

// Network fee quotes, exactly as FEES.md section 5 prescribes:
//  1. build the transaction, sign a copy with a throwaway key and ask the DogeOS oracle for getL1Fee (a dummy-signed copy
//     quotes within 0.3% of the real one), add gasLimit x maxFeePerGas (negligible): that is the estimate;
//  2. show a range: "about <estimate>, up to <high>" with high = estimate x 1.5 (covers the largest 30 s rise seen, +33.6%);
//  3. if the oracle call fails, fall back to the generous fixed ceilings of section 5.5.
export type FeeAction = "tip" | "claim" | "refund";

/** Gas limits used only to build the quote (gasUsed measured in FEES.md 2b: tip 136,466, claim 118,752, refund 44,461). */
const QUOTE_GAS: Record<FeeAction, bigint> = { tip: 140_000n, claim: 125_000n, refund: 50_000n };
/** FEES.md 5.5: used when the oracle cannot be read. Shown as "up to". */
export const FALLBACK_CEILING: Record<FeeAction, bigint> = { tip: parseEther("0.35"), claim: parseEther("0.85"), refund: parseEther("0.30") };

export interface FeeQuote {
  /** The oracle estimate in wei, or null when the fallback ceiling is used. */
  aboutWei: bigint | null;
  /** The buffered upper bound: estimate x 1.5, or the fallback ceiling. Balance checks use this. */
  upToWei: bigint;
  source: "oracle" | "fallback";
}

export const buffered = (estimate: bigint) => (estimate * 3n) / 2n;

/** Representative calldata for each action (the L1 fee depends on the transaction's size, not on the exact values). */
function sampleCall(action: FeeAction): { data: Hex; value: bigint } {
  if (action === "tip") return { data: encodeFunctionData({ abi: escrowAbi, functionName: "tip", args: [QUOTE_SAMPLE.idHash] }), value: parseEther(QUOTE_SAMPLE.tipDoge) };
  if (action === "refund") return { data: encodeFunctionData({ abi: escrowAbi, functionName: "refund", args: [1n] }), value: 0n };
  return {
    data: encodeFunctionData({
      abi: escrowAbi,
      functionName: "claim",
      args: [QUOTE_SAMPLE.claim.xId, QUOTE_SAMPLE.claim.recipient, QUOTE_SAMPLE.claim.nonce, QUOTE_SAMPLE.claim.expiry, QUOTE_SAMPLE.claim.signature],
    }),
    value: 0n,
  };
}

/**
 * Quote the network fee for a call. With `from` (the connected wallet) the quoted transaction is the real one: its
 * nonce and an estimateGas for that sender (falling back to the measured gas when estimation fails, e.g. the wallet
 * can't afford the value yet). Without `from`, nonce 0 and the measured gas: the L1 fee depends on the transaction's
 * size, which those barely change.
 */
export async function quoteFee(client: PublicClient, action: FeeAction, call?: { data: Hex; value: bigint }, from?: Address): Promise<FeeQuote> {
  try {
    const { data, value } = call ?? sampleCall(action);
    const [fees, nonce, gas] = await Promise.all([
      client.estimateFeesPerGas(),
      from ? client.getTransactionCount({ address: from, blockTag: "pending" }) : Promise.resolve(0),
      from ? client.estimateGas({ account: from, to: ESCROW_ADDRESS, data, value }).catch(() => QUOTE_GAS[action]) : Promise.resolve(QUOTE_GAS[action]),
    ]);
    const throwaway = privateKeyToAccount(generatePrivateKey()); // never stored, never funded
    const signed = await throwaway.signTransaction({
      chainId: CHAIN_ID,
      type: "eip1559",
      to: ESCROW_ADDRESS,
      data,
      value,
      gas,
      nonce,
      maxFeePerGas: fees.maxFeePerGas,
      maxPriorityFeePerGas: fees.maxPriorityFeePerGas,
    });
    const l1 = await client.readContract({ address: L1_GAS_ORACLE, abi: oracleAbi, functionName: "getL1Fee", args: [signed] });
    const estimate = l1 + gas * fees.maxFeePerGas;
    return { aboutWei: estimate, upToWei: buffered(estimate), source: "oracle" };
  } catch {
    return { aboutWei: null, upToWei: FALLBACK_CEILING[action], source: "fallback" };
  }
}

/** Wiring-spec W2.6: the tip may go ahead only if the balance covers the amount plus the buffered fee. */
export const coversTip = (balanceWei: bigint, amountWei: bigint, feeHighWei: bigint) => balanceWei >= amountWei + feeHighWei;

/**
 * "Tip N DOGE instead": N = balance - feeHigh, rounded down to whole DOGE, offered only if it is at least the minimum
 * tip. Returns N in whole DOGE, or null.
 */
export function affordableTip(balanceWei: bigint, feeHighWei: bigint, minTipWei: bigint): bigint | null {
  if (balanceWei <= feeHighWei) return null;
  const whole = (balanceWei - feeHighWei) / 10n ** 18n;
  return whole * 10n ** 18n >= minTipWei && whole > 0n ? whole : null;
}

/** How often to refresh a quote while a form is open (FEES.md 5.2, wiring-spec W2.4: every 15 s). */
export const FEE_REFRESH_MS = 15_000;
```

### `src/data/api.ts`

```ts
import { encodeAbiParameters, keccak256 } from "viem";
import { ENV } from "../env";
import { OFFLINE } from "../chain/errors";
import { DEMO_ACCOUNTS } from "./samples";
import type { ApiErrorBody, FeedResponse, ResolvedHandle, SignedClaim } from "./types";

// The Supabase Edge Functions of docs/backend.md, configured with Vite env vars (src/env.ts, frontend/.env.example).
// When they aren't set the backend isn't connected. In development the app then uses built-in fixtures, always labelled
// "Demo" / "Example data", so the UI can be worked on; production builds never show sample data (wiring-spec W4.2) and
// report the backend as unreachable instead. Sign-in and real tips are off either way.
export const SUPABASE_URL: string | undefined = ENV.supabaseUrl ?? undefined;
export const SUPABASE_ANON_KEY: string | undefined = ENV.supabaseAnonKey ?? undefined;
export const FUNCTIONS_URL: string | undefined = ENV.functionsUrl ?? undefined;
export const backendConfigured = Boolean(FUNCTIONS_URL);
/** Sign in with X needs the Supabase project itself (Auth), not only the functions. */
export const authConfigured = Boolean(SUPABASE_URL && SUPABASE_ANON_KEY);
/** Labelled demo fixtures stand in for the backend: development builds only. */
export const demoMode = !backendConfigured && import.meta.env.DEV;

export class ApiError extends Error {
  constructor(public status: number, public code: string, message: string, public path = "") {
    super(message);
  }
}

async function call<T>(path: string, init: RequestInit = {}): Promise<T> {
  if (!FUNCTIONS_URL) throw new ApiError(0, "backend_not_configured", OFFLINE);
  const headers = new Headers(init.headers);
  if (SUPABASE_ANON_KEY && !headers.has("apikey")) headers.set("apikey", SUPABASE_ANON_KEY);
  let res: Response;
  try {
    res = await fetch(`${FUNCTIONS_URL}/${path}`, { ...init, headers });
  } catch {
    throw new ApiError(0, "network", OFFLINE);
  }
  const body = (await res.json().catch(() => null)) as T | ApiErrorBody | null;
  if (!res.ok || !body || (typeof body === "object" && "error" in body)) {
    const err = (body as ApiErrorBody | null)?.error;
    throw new ApiError(res.status, err?.code ?? "bad_response", err?.message ?? `The backend answered ${res.status}`, path.split("?")[0]);
  }
  return body as T;
}

/** Same rule as the backend: trim, drop one leading @, 1 to 15 of A-Z a-z 0-9 _. */
export const HANDLE_RE = /^[A-Za-z0-9_]{1,15}$/;
/**
 * What the user typed or pasted -> a handle to validate: trims, drops one leading @, and turns a pasted profile URL
 * (x.com/<handle>, twitter.com/<handle>, with or without https:// and www., trailing path or query) into its handle
 * (wiring-spec W2.2). The backend still rejects URLs; this only happens in the UI.
 */
export function normalizeHandle(raw: string) {
  const s = raw.trim();
  const url = s.match(/^(?:https?:\/\/)?(?:www\.|mobile\.)?(?:x|twitter)\.com\/@?([A-Za-z0-9_]{1,15})(?:[/?#].*)?$/i);
  return url ? url[1] : s.replace(/^@/, "");
}

/**
 * Wiring-spec section 6: what to show for a backend error code. Only handle_not_found says "No X account called @…";
 * a network or CORS failure, X being down, rate limits, a missing backend config and a demo miss each have their own.
 * Unknown codes fall back to the server's own message (they are written for people).
 */
export function apiErrorMessage(e: unknown, ctx: { handle?: string } = {}): string {
  if (!(e instanceof ApiError)) return OFFLINE;
  switch (e.code) {
    case "handle_not_found":
      return `No X account called @${ctx.handle ?? "that handle"}.`;
    case "invalid_handle":
      return "Handles are 1 to 15 letters, numbers or underscores.";
    case "upstream_rate_limited":
      return "Too many lookups. Try again in a minute.";
    case "rate_limited":
      return e.path === "sign-claim" ? "Too many claim attempts. Try again in a few minutes." : "Too many lookups. Try again in a minute.";
    case "upstream_unavailable":
      return "We can't reach X right now. Try again shortly.";
    case "invalid_token":
    case "missing_token":
      return "Your X sign-in expired. Sign in again.";
    case "identity_not_allowed":
      return "This account is linked to something other than X. Sign out and sign in with X only.";
    case "chain_unavailable":
    case "wrong_chain":
    case "verifier_mismatch":
      return "Claiming is unavailable right now. Your tips are safe in the contract.";
    case "network":
      return OFFLINE; // also what a CORS rejection looks like from the browser
    case "backend_not_configured":
      return "Handle lookups aren't available on this site yet: its backend isn't configured.";
    case "demo_unknown":
      return `Demo mode: @${ctx.handle ?? "that handle"} isn't one of the demo accounts. Nothing was checked with X.`;
    default:
      return e.message || OFFLINE;
  }
}

/** TipEscrow.idHash(xId): keccak256 of the X id as one 32-byte word. */
export const idHashOf = (xId: string | bigint) => keccak256(encodeAbiParameters([{ type: "uint256" }], [BigInt(xId)]));


/** X handle -> numeric X id. `fresh` always asks X (the tip flow must use it right before confirming). null = not found. */
export async function resolveHandle(handle: string, opts: { fresh?: boolean; signal?: AbortSignal } = {}): Promise<ResolvedHandle | null> {
  const h = normalizeHandle(handle);
  if (demoMode) {
    await new Promise((r) => setTimeout(r, 300));
    const a = DEMO_ACCOUNTS[h.toLowerCase()];
    // A handle outside the demo list is NOT "no such X account": nothing was asked of X (see apiErrorMessage)
    if (!a) throw new ApiError(0, "demo_unknown", "Not a demo account", "resolve");
    return { ...a, idHash: idHashOf(a.xId), source: "demo" };
  }
  const q = new URLSearchParams({ handle: h });
  if (opts.fresh) q.set("fresh", "1");
  try {
    return await call<ResolvedHandle>(`resolve?${q}`, { signal: opts.signal });
  } catch (e) {
    if (e instanceof ApiError && e.code === "handle_not_found") return null;
    throw e;
  }
```

```ts
// Display helpers. Amounts are wei (bigint or decimal string) and are formatted with BigInt only, never floats.

const WEI = 10n ** 18n;
const group = (s: string) => s.replace(/\B(?=(\d{3})+(?!\d))/g, ",");

/** Wei -> DOGE with at most `maxDecimals` decimals (truncated), trailing zeros dropped, thousands grouped. */
export function formatDoge(wei: string | bigint, maxDecimals = 4): string {
  const v = typeof wei === "bigint" ? wei : BigInt(wei);
  const neg = v < 0n;
  const abs = neg ? -v : v;
  const frac = (abs % WEI).toString().padStart(18, "0").slice(0, maxDecimals).replace(/0+$/, "");
  return `${neg ? "-" : ""}${group((abs / WEI).toString())}${frac ? "." + frac : ""}`;
}

/** Wei -> DOGE with exactly `decimals` decimals, rounded up or down. "At most" figures round up so they stay true. */
export function formatDogeFixed(wei: bigint, decimals: number, round: "up" | "down" = "down"): string {
  const unit = 10n ** BigInt(18 - decimals);
  const scaled = round === "up" ? (wei + unit - 1n) / unit : wei / unit;
  const s = scaled.toString().padStart(decimals + 1, "0");
  const whole = s.slice(0, s.length - decimals);
  return decimals ? `${group(whole)}.${s.slice(-decimals)}` : group(whole);
}

/**
 * "0.02 – 0.03" from an estimate and its upper bound, the bound rounded up. Two decimals as in the spec, more only when
 * the estimate would otherwise read "0.00" (fees on Chikyū can drop below a cent).
 */
export function formatFeeRange(aboutWei: bigint | null, upToWei: bigint): string {
  if (aboutWei === null) return `up to ${formatDogeFixed(upToWei, 2, "up")}`;
  let d = 2;
  while (d < 6 && formatDogeFixed(aboutWei, d, "down").replace(/[0.]/g, "") === "") d++;
  const lo = formatDogeFixed(aboutWei, d, "down");
  const hi = formatDogeFixed(upToWei, d, "up");
  return lo === hi ? `≈ ${hi}` : `${lo} – ${hi}`;
}

/** The app's fee wording: "0.01 to 0.02" (spec A1), "up to 0.35" for the fallback ceiling. */
export const formatFeeTo = (aboutWei: bigint | null, upToWei: bigint) => formatFeeRange(aboutWei, upToWei).replace(" – ", " to ");

export const shortAddr = (a: string) => `${a.slice(0, 6)}…${a.slice(-4)}`;
export const shortHash = (h: string) => `${h.slice(0, 6)}…${h.slice(-4)}`;

const clock = (d: Date) => d.toLocaleTimeString("en-GB", { hour: "2-digit", minute: "2-digit" });

/** "Oct 31, 13:24" in the viewer's time zone (activity rows). */
export function formatEventTime(iso: string | Date): string {
  const d = typeof iso === "string" ? new Date(iso) : iso;
  return `${formatDay(d)}, ${clock(d)}`;
}

/** "Today 13:24", "Yesterday", "Oct 31" (My tips rows). */
export function formatDayLabel(d: Date, now = new Date()): string {
  if (d.toDateString() === now.toDateString()) return `Today ${clock(d)}`;
  const y = new Date(now);
  y.setDate(now.getDate() - 1);
  return d.toDateString() === y.toDateString() ? "Yesterday" : formatDay(d);
}

/** "13:24" */
export const formatClockTime = (d: Date) => clock(d);

/** "Oct 30" */
export const formatDay = (d: Date) => d.toLocaleDateString("en-US", { month: "short", day: "numeric" });

/** "13:34 today" / "Oct 31 · 13:34" */
export function formatClock(d: Date, now = new Date()): string {
  const time = d.toLocaleTimeString("en-GB", { hour: "2-digit", minute: "2-digit" });
  return d.toDateString() === now.toDateString() ? `${time} today` : `${formatDay(d)} · ${time}`;
}

export function relativeTime(iso: string | Date, now = Date.now()): string {
  const t = typeof iso === "string" ? Date.parse(iso) : iso.getTime();
  const s = Math.max(0, Math.round((now - t) / 1000));
  if (s < 30) return "just now";
  if (s < 3600) return `${Math.max(1, Math.round(s / 60))} min ago`;
  if (s < 86400) return `${Math.floor(s / 3600)} h ago`;
  return `${Math.floor(s / 86400)} d ago`;
}

/** 600 -> "10 min", 90 -> "90 s", 7200 -> "2 h", 604800 -> "7 days". */
export function formatDuration(seconds: number): string {
  if (seconds < 120) return `${seconds} s`;
  if (seconds < 7200) return `${Math.round(seconds / 60)} min`;
  if (seconds < 172800) return `${Math.round(seconds / 3600)} h`;
  return `${Math.round(seconds / 86400)} days`;
}

/** 600 -> "10 minutes", 60 -> "1 minute", 7200 -> "2 hours", 604800 -> "7 days" (sentence copy). */
export function formatDurationLong(seconds: number): string {
  const unit = (n: number, w: string) => `${n} ${w}${n === 1 ? "" : "s"}`;
  if (seconds < 120 && seconds % 60 !== 0) return unit(seconds, "second");
  if (seconds < 7200) return unit(Math.round(seconds / 60), "minute");
  if (seconds < 172800) return unit(Math.round(seconds / 3600), "hour");
  return unit(Math.round(seconds / 86400), "day");
}

/** Seconds left -> "6:02" under an hour, "2:05:09" under two days, else "3 days". */
export function formatCountdown(seconds: number): string {
  const s = Math.max(0, Math.floor(seconds));
  if (s >= 172800) return `${Math.floor(s / 86400)} days`;
  const h = Math.floor(s / 3600);
  const m = Math.floor((s % 3600) / 60);
  const ss = String(s % 60).padStart(2, "0");
  return h ? `${h}:${String(m).padStart(2, "0")}:${ss}` : `${m}:${ss}`;
}

/** Initials for an avatar: "X Developers" -> "XD", "jack" -> "JA". */
export function initials(name: string | null | undefined, handle: string): string {
  const words = (name || "").replace(/[^\p{L}\p{N} ]/gu, " ").trim().split(/\s+/).filter(Boolean);
  if (words.length >= 2) return (words[0][0] + words[1][0]).toUpperCase();
  return (words[0] || handle).slice(0, 2).toUpperCase();
}
```

```ts
/** Indexed tips + their on-chain status -> the rows the UI shows. Tips the chain doesn't know (status None) are dropped. */
export function withStatus(rows: IndexedTip[], statuses: readonly unknown[] | undefined): OnchainTip[] | undefined {
  if (!rows.length) return [];
  if (!statuses) return undefined;
  const out: OnchainTip[] = [];
  rows.forEach((r, k) => {
    const status = tipStatusOf(statuses[2 * k] as number);
    if (!status) return;
    out.push({ id: r.tipId, idHash: r.idHash, tipper: r.tipper, createdAt: r.time, amountWei: r.amountWei, status, refundableAt: Number(statuses[2 * k + 1] as bigint) });
  });
  return out;
}

/**
 * Tips for one X account (by idHash) or from one tipper, straight from the contract's storage: tips(i) and tipStatus(i)
 * for the most recent TIP_SCAN_LIMIT tips, batched through Multicall3. This is the user's own fact, so it is read from the
 * chain rather than the indexer (docs/backend.md, "Why feed is server-side").
 */
export function useTips(filter: { idHash?: Hex; tipper?: Address }, refundDelaySeconds?: number) {
  const client = usePublicClient({ chainId: CHAIN_ID });
  const enabled = Boolean(client && (filter.idHash || filter.tipper) && refundDelaySeconds !== undefined);
  return useQuery({
    queryKey: ["tips", filter.idHash ?? null, filter.tipper?.toLowerCase() ?? null],
    enabled,
    refetchInterval: 15_000,
    queryFn: async (): Promise<OnchainTip[]> => {
      const c = client as PublicClient;
      const count = await c.readContract({ ...escrow, functionName: "tipCount" });
      if (count === 0n) return [];
      const from = count > TIP_SCAN_LIMIT ? count - TIP_SCAN_LIMIT + 1n : 1n;
      const ids: bigint[] = [];
      for (let i = count; i >= from; i--) ids.push(i);
      const res = await c.multicall({
        allowFailure: false,
        contracts: ids.flatMap((id) => [
          { ...escrow, functionName: "tips", args: [id] } as const,
          { ...escrow, functionName: "tipStatus", args: [id] } as const,
        ]),
      });
      const out: OnchainTip[] = [];
      ids.forEach((id, k) => {
        const [idHash, tipper, createdAt, , amount] = res[2 * k] as readonly [Hex, Address, bigint, boolean, bigint];
        const status = tipStatusOf(res[2 * k + 1] as number);
        if (!status) return;
        if (filter.idHash && idHash.toLowerCase() !== filter.idHash.toLowerCase()) return;
        if (filter.tipper && tipper.toLowerCase() !== filter.tipper.toLowerCase()) return;
        out.push({ id, idHash, tipper, createdAt: Number(createdAt), amountWei: amount, status, refundableAt: Number(createdAt) + (refundDelaySeconds ?? 0) });
      });
      return out;
    },
  });

/**
 * Contract events filtered by an indexed argument, read from the chain in 9,000-block chunks (DogeOS refuses ranges over
 * 10,000) from the deploy block to the head, newest first, with block times.
 */
async function readLogs(client: PublicClient, eventName: "Tipped" | "Claimed" | "Refunded", args: Record<string, unknown>): Promise<ChainLog[]> {
  const head = await client.getBlockNumber();
  const ranges: [bigint, bigint][] = [];
  for (let from = ESCROW_DEPLOY_BLOCK; from <= head; from += LOG_RANGE_BLOCKS) {
    const to = from + LOG_RANGE_BLOCKS - 1n;
    ranges.push([from, to > head ? head : to]);
  }
  const chunks = await Promise.all(
    ranges.map(([fromBlock, toBlock]) =>
      client.getContractEvents({ address: ESCROW_ADDRESS, abi: escrowAbi, eventName, args: args as never, fromBlock, toBlock, strict: true }),
    ),
  );
  const logs = chunks.flat();
  const blocks = [...new Set(logs.map((l) => l.blockNumber))];
  const times = new Map<bigint, number>();
  await Promise.all(blocks.map(async (b) => times.set(b, Number((await client.getBlock({ blockNumber: b })).timestamp))));
  return logs
    .map((l) => ({ txHash: l.transactionHash, blockNumber: l.blockNumber, time: times.get(l.blockNumber) ?? 0, args: l.args as Record<string, unknown> }))
    .sort((a, b) => Number(b.blockNumber - a.blockNumber));
}

    enabled: Boolean(client && args),
    refetchInterval: 20_000,
    queryFn: () => readLogs(client as PublicClient, eventName, args ?? {}),
  });
}

const sumPending = (tips: OnchainTip[]) => tips.reduce((s, t) => s + t.amountWei, 0n);

/**
 * W3.3 with the backend: the tips from chain_events (event Tipped, this idHash), each one's tipStatus and refundableAt
 * read from the chain in one batch; "Claimed so far" from chain_events Claimed; the total from balanceOf.
 */
function useClaimSummaryIndexed(idHash: Hex | null): ClaimSummary {
  const ready = useReadContract({ ...escrow, functionName: "balanceOf", args: idHash ? [idHash] : undefined, query: { enabled: Boolean(idHash), refetchInterval: 15_000 } });
  const listed = useQuery({ queryKey: ["indexed-tips-for", idHash], queryFn: () => tipsFor(idHash as Hex), enabled: Boolean(idHash), refetchInterval: 15_000 });
  const claims = useQuery({ queryKey: ["indexed-claims-for", idHash], queryFn: () => claimsFor(idHash as Hex), enabled: Boolean(idHash), refetchInterval: 15_000 });
  const rows = listed.data ?? [];
  const statuses = useTipStatuses(rows.map((r) => r.tipId));
  const tips = withStatus(rows, statuses.data);
  const pending = (tips ?? []).filter((t) => t.status === "Pending").sort((a, b) => a.refundableAt - b.refundableAt);
  const readyWei = ready.data as bigint | undefined;
  return {
    readyWei,
    pending,
    loading: readyWei === undefined || listed.isPending || tips === undefined,
    claimedWei: claims.data ? claims.data.reduce((s, c) => s + c.amountWei, 0n) : undefined,
    claimedCount: (tips ?? []).filter((t) => t.status === "Claimed").length,
    txOfTip: new Map(rows.map((r) => [r.tipId.toString(), r.txHash])),
    syncing: readyWei !== undefined && tips !== undefined && readyWei > sumPending(pending),
    refetch: () => Promise.all([ready.refetch(), listed.refetch(), claims.refetch(), statuses.refetch()]),
  };
}

/** Without the backend: the same facts straight from the contract (tips(i) scan and Claimed logs). */
function useClaimSummaryFromChain(idHash: Hex | null): ClaimSummary {
  const { refundDelaySeconds } = useEscrowConfig();
  const ready = useReadContract({ ...escrow, functionName: "balanceOf", args: idHash ? [idHash] : undefined, query: { enabled: Boolean(idHash), refetchInterval: 15_000 } });
  const tips = useTips({ idHash: idHash ?? undefined }, refundDelaySeconds);
  const claims = useEscrowLogs("Claimed", idHash ? { idHash } : null);
  const tipped = useEscrowLogs("Tipped", idHash ? { idHash } : null);
  const list = tips.data ?? [];
  const pending = list.filter((t) => t.status === "Pending").sort((a, b) => a.refundableAt - b.refundableAt);
  const readyWei = ready.data as bigint | undefined;
  return {
    readyWei,
    pending,
    loading: readyWei === undefined || tips.isPending,
    claimedWei: claims.data ? claims.data.reduce((s, c) => s + (c.args.amount as bigint), 0n) : undefined,
    claimedCount: list.filter((t) => t.status === "Claimed").length,
    txOfTip: new Map((tipped.data ?? []).map((l) => [(l.args.tipId as bigint).toString(), l.txHash])),
    syncing: readyWei !== undefined && !tips.isPending && readyWei > sumPending(pending),
    refetch: () => Promise.all([ready.refetch(), tips.refetch(), claims.refetch()]),
  };
}

/** Everything the Claim tab shows for one X account (W3.3). Indexed when the backend is configured, else from the chain. */
export const useClaimSummary = eventsConfigured ? useClaimSummaryIndexed : useClaimSummaryFromChain;

/** The four rows of My tips (W4.1), from the chain's status and the refund time. */
export type TipPillKind = "waiting" | "refundable" | "claimed" | "refunded";
export function tipPillOf(t: Pick<OnchainTip, "status" | "refundableAt">, now: number): TipPillKind {
  if (t.status === "Refunded") return "refunded";
  if (t.status === "Claimed") return "claimed";
  return now >= t.refundableAt ? "refundable" : "waiting";
}

export interface SentTips {
  /** undefined while loading (or with no wallet) */
  tips: OnchainTip[] | undefined;
  /** idHash (lowercase) -> handle, from x_handles */
  handles: Map<string, string>;
  txOfTip: Map<string, Hex>;
  hasMore: boolean;
  loadMore: () => void;
  loadingMore: boolean;
  refetch: () => Promise<unknown>;

/** W4.1: chain_events (Tipped, account = this wallet), 20 per page, newest first; handles from x_handles; status from the chain. */
function useSentTipsIndexed(address: Address | undefined): SentTips {
  const q = useInfiniteQuery({
    queryKey: ["indexed-tips-by", address?.toLowerCase() ?? null],
    queryFn: ({ pageParam }) => tipsBy(address as Address, pageParam),
    initialPageParam: 0,
    getNextPageParam: (last, pages) => (last.hasMore ? pages.length : undefined),
    enabled: Boolean(address),
    refetchInterval: 15_000,
  });
  const rows = q.data?.pages.flatMap((p) => p.rows) ?? [];
  const statuses = useTipStatuses(rows.map((r) => r.tipId));
  const ids = [...new Set(rows.map((r) => r.idHash.toLowerCase()))].sort();
  const names = useQuery({ queryKey: ["x-handles", ids.join(",")], queryFn: () => handlesFor(ids), enabled: ids.length > 0, staleTime: 60_000 });
  return {
    tips: address && q.data ? withStatus(rows, statuses.data) : undefined,
    handles: names.data ?? new Map(),
    txOfTip: new Map(rows.map((r) => [r.tipId.toString(), r.txHash])),
    hasMore: Boolean(q.hasNextPage),
    loadMore: () => void q.fetchNextPage(),
    loadingMore: q.isFetchingNextPage,
    refetch: () => Promise.all([q.refetch(), statuses.refetch()]),
  };
}

/** Without the backend: the wallet's tips straight from the contract (tips(i) scan, Tipped logs for the tx hashes). */

    handles: new Map(),
    txOfTip: new Map((sent.data ?? []).map((l) => [(l.args.tipId as bigint).toString(), l.txHash])),
    hasMore: false,
    loadMore: () => undefined,
    loadingMore: false,

  };

export type SetupStepState = "done" | "next" | "todo";

/**
 * The three setup steps (wiring-spec W1.8): 1 a wallet is connected, 2 it is on Chikyū, 3 its balance covers the minimum
 * tip plus the tip's buffered network fee (minTip + tipFeeHigh), i.e. it can actually send a tip.
 * `known` is false until the numbers have loaded, so the setup strip doesn't flash for a funded wallet.
 */
export function useSetupSteps() {
  const wallet = useWallet();
  const { minTipWei } = useEscrowConfig();
  const hasWallet = wallet.status === "connected" || wallet.status === "wrong-network";
  const fee = useFeeQuote("tip", undefined, hasWallet);
  const needWei = minTipWei !== undefined && fee.data ? minTipWei + fee.data.upToWei : undefined;
  const funded = hasWallet && wallet.balanceWei !== undefined && needWei !== undefined && wallet.balanceWei >= needWei;
  const done = [hasWallet, wallet.status === "connected", funded];
  const next = done.indexOf(false);
  const states: SetupStepState[] = done.map((d, i) => (d ? "done" : i === next ? "next" : "todo"));
  return {
    states,
    doneCount: done.filter(Boolean).length,
    allDone: next === -1,
    needWei,
    known: wallet.status === "disconnected" || (hasWallet && wallet.balanceWei !== undefined && needWei !== undefined),
  };
}
```

```ts
import { decodeFunctionData, parseEventLogs, type Hex, type PublicClient, type TransactionReceipt } from "viem";
import { escrowAbi } from "./abi";
import { ESCROW_ADDRESS } from "./config";
import { describeTxError, type ErrorContext, type TxError } from "./errors";

/**
 * Fee actually paid, from the RAW receipt: l1Fee + gasUsed x effectiveGasPrice (FEES.md 1 and 5.6). viem's receipt
 * object drops l1Fee, so the RPC method is called directly.
 */
export async function feePaid(client: PublicClient, hash: Hex): Promise<bigint | null> {
  try {
    const raw = (await client.request({ method: "eth_getTransactionReceipt", params: [hash] } as never)) as { l1Fee?: Hex; gasUsed: Hex; effectiveGasPrice: Hex } | null;
    if (!raw) return null;
    return BigInt(raw.l1Fee ?? "0x0") + BigInt(raw.gasUsed) * BigInt(raw.effectiveGasPrice);
  } catch {
    return null;
  }
}

/** The tipId of a mined tip, from its Tipped log. */
export function tipIdOf(receipt: TransactionReceipt): bigint | null {
  const logs = parseEventLogs({ abi: escrowAbi, eventName: "Tipped", logs: receipt.logs.filter((l) => l.address.toLowerCase() === ESCROW_ADDRESS.toLowerCase()) });
  return logs[0]?.args.tipId ?? null;
}

/**
 * Why a mined transaction reverted: replays the same call (same sender, calldata and value) at the block before it was
 * included, which gives the contract's custom error (e.g. TooEarly, NothingToClaim). Best effort: if the replay
 * succeeds (the state changed in that block) the error is generic.
 */
export async function revertReason(client: PublicClient, hash: Hex, blockNumber: bigint, ctx: ErrorContext = {}): Promise<TxError> {
  const generic: TxError = { kind: "revert", title: "The contract refused it", detail: "The transaction was mined but the contract reverted it. Only the network fee was spent." };
  try {
    const tx = await client.getTransaction({ hash });
    const { functionName, args } = decodeFunctionData({ abi: escrowAbi, data: tx.input });
    await client.simulateContract({ address: ESCROW_ADDRESS, abi: escrowAbi, functionName, args, value: tx.value, account: tx.from, blockNumber: blockNumber - 1n } as never);
    return generic;
  } catch (e) {
    const d = describeTxError(e, ctx);
    return d.kind === "revert" ? d : generic;
  }
}
```

### `src/lib/authError.ts`

```ts
// A failed "Sign in with X" comes back to /app/claim with the reason in the URL: in the query
// (?error=access_denied&error_code=…&error_description=…) or, for the implicit flow, in the hash (#error=…).

export interface AuthRedirectError {
  /** e.g. "access_denied", "server_error" */
  code: string;
  /** X's or Supabase's own explanation, if any (untrusted text: render as text only) */
  reason: string | null;
}

const KEYS = ["error", "error_code", "error_description"];

const params = (s: string) => new URLSearchParams(s.replace(/^[?#]/, ""));

/** The sign-in error carried by a URL's query or hash, or null. The query wins if both have one. */
export function readAuthError(search: string, hash: string): AuthRedirectError | null {
  for (const p of [params(search), params(hash)]) {
    const error = p.get("error");
    const description = p.get("error_description");
    if (!error && !description) continue;
    const reason = description?.trim().slice(0, 200) || null;
    return { code: p.get("error_code") || error || "unknown_error", reason };
  }
  return null;
}

/** The same URL without the error parameters (everything else, e.g. a sign-in code, is kept). */
export function withoutAuthError(pathname: string, search: string, hash: string): string {
  // only a part that carries an error key is rewritten, so a plain anchor like #faq survives untouched
  const strip = (s: string) => {
    const p = params(s);
    if (!KEYS.some((k) => p.has(k))) return s.replace(/^[?#]/, "");
    KEYS.forEach((k) => p.delete(k));
    return p.toString();
  };
  const q = strip(search);
  const h = strip(hash);
  return `${pathname}${q ? `?${q}` : ""}${h ? `#${h}` : ""}`;
}

export const CLAIM_PATH = "/app/claim";

/**
 * Supabase sends a failed sign-in to the Site URL (or wherever it was started), e.g. /?error=…. Only the Claim page
 * shows the error, so any other page carrying one is sent there with its query and hash unchanged. null = stay.
 */
export function authErrorTarget(pathname: string, search: string, hash: string): string | null {
  if (pathname === CLAIM_PATH || !readAuthError(search, hash)) return null;
  return `${CLAIM_PATH}${search}${hash}`;
}
```

```tsx
/** A connect request the wallet never answers (locked, popup closed) stops waiting after this long. */
export const CONNECT_TIMEOUT_MS = 30_000;

export class ConnectTimeout extends Error {
  constructor() {
    super("The wallet didn't answer");
  }
}

/**
 * What to tell the user when connecting fails. Every case ends in a message and a retry, never a spinner:
 * 4001 rejected, -32002 a request already open in the wallet, a wallet that never answered (locked or popup closed).
 */
export function connectErrorMessage(e: unknown): string {
  if (e instanceof ConnectTimeout) return "Your wallet didn't answer. It may be locked or waiting in its window: open the wallet extension, unlock it, then try again.";
  const err = e as { code?: number; cause?: { code?: number }; message?: string; shortMessage?: string } | null;
  const code = err?.code ?? err?.cause?.code;
  const text = `${err?.shortMessage ?? ""} ${err?.message ?? ""}`;
  if (code === -32002 || /already pending/i.test(text)) return "Your wallet already has a connection request open. Open the wallet extension, approve or close it, then try again.";
  if (code === 4001 || /reject|denied|cancel/i.test(text)) return "You cancelled in your wallet.";
  return "Couldn't connect the wallet. Try again.";
}

/** Rejects with ConnectTimeout if `p` hasn't settled after `ms`. */
export function withTimeout<T>(p: Promise<T>, ms: number): Promise<T> {
  let t: ReturnType<typeof setTimeout> | undefined;
  return Promise.race([p, new Promise<never>((_, reject) => (t = setTimeout(() => reject(new ConnectTimeout()), ms)))]).finally(() => clearTimeout(t));
}

type ConnectState = { kind: "idle" } | { kind: "waiting" } | { kind: "error"; message: string };

/**
 * One connect attempt with a time limit. The UI follows this state, not wagmi's isPending (which stays true forever
 * when a wallet never answers). If the wallet does answer late, wagmi still connects and the UI follows the account.
 */
function useConnectFlow() {
  const { connectAsync } = useConnect();
  const [state, setState] = useState<ConnectState>({ kind: "idle" });
  const attempt = useRef(0);
  const connect = async (c: Connector | undefined): Promise<{ ok: true } | { ok: false; message: string }> => {
    const id = ++attempt.current;
    if (!c) {
      setState({ kind: "error", message: NO_WALLET });
      return { ok: false, message: NO_WALLET };
    }
    setState({ kind: "waiting" });
    try {
      await withTimeout(connectAsync({ connector: c }), CONNECT_TIMEOUT_MS);
      if (id === attempt.current) setState({ kind: "idle" });
      return { ok: true };
    } catch (e) {
      const message = connectErrorMessage(e);
      if (id === attempt.current) setState({ kind: "error", message });
      return { ok: false, message };
    }
  };
  return { state, connect, dismiss: () => setState({ kind: "idle" }) };
}

/** Connect for a call-to-action outside the header: the only wallet, or the browser's default (window.ethereum) one. */
export function useConnectWallet() {
  const { connectors } = useConnect();
  const wallets = useWallets();
  const toast = useToast();
  const flow = useConnectFlow();
  const connect = async (): Promise<boolean> => {
    const hasDefault = typeof window !== "undefined" && "ethereum" in window;

    const r = await flow.connect(wallets.length ? c : undefined);

    return r.ok;
  };
  return { connect, isConnecting: flow.state.kind === "waiting" };
}

function useOutside(open: boolean, close: () => void) {
  const ref = useRef<HTMLDivElement>(null);
  useEffect(() => {
    if (!open) return;
    const onDown = (e: MouseEvent) => ref.current && !ref.current.contains(e.target as Node) && close();
    const onKey = (e: KeyboardEvent) => e.key === "Escape" && close();
    document.addEventListener("mousedown", onDown);
    document.addEventListener("keydown", onKey);
    return () => {
      document.removeEventListener("mousedown", onDown);
      document.removeEventListener("keydown", onKey);
    };
  }, [open, close]);
  return ref;
}
```

```ts
import { useQuery } from "@tanstack/react-query";
import { useEffect, useState } from "react";
import { apiErrorMessage, backendConfigured, demoMode, fetchFeed, HANDLE_RE, normalizeHandle, resolveHandle } from "./api";
import { handleFor } from "../lib/handleBook";

import { demoFeed } from "./samples";
import type { EventType, FeedResponse, ResolvedHandle } from "./types";

export type FeedFilter = "All" | EventType;
export type FeedState =
  | { kind: "loading" }
  | { kind: "error"; lastSync: number | null; retry: () => void; data: FeedResponse | null }
  | { kind: "ready"; data: FeedResponse; source: "live" | "demo"; /** the index is behind the chain (sync.complete false) */ syncing: boolean };

/** Wiring-spec W4.2: poll every 5 s; react-query pauses the interval while the tab is hidden. */
export const FEED_POLL_MS = 5_000;

/**
 * The activity feed (W4.2): feed?limit=&type= every 5 s while the tab is visible. States: loading (F2), ready (F1, or
 * F3 when there are no events), error (F4), which keeps the last good rows in `data` with the time of the last
 * successful update. In development without a backend the labelled demo rows stand in; production builds never show
 * them, so there the feed is simply unreachable until the backend is configured.
 */
export function useFeed(filter: FeedFilter, limit = 25): FeedState {
  const q = useQuery({
    queryKey: ["feed", filter, limit],
    queryFn: ({ signal }) => fetchFeed({ limit, type: filter === "All" ? undefined : filter, signal }),
    enabled: backendConfigured,
    refetchInterval: FEED_POLL_MS,
    refetchIntervalInBackground: false,
    retry: 1,
  });
  if (demoMode) {
    const events = filter === "All" ? demoFeed.events : demoFeed.events.filter((e) => e.type === filter);
    return { kind: "ready", data: { ...demoFeed, events: events.slice(0, limit) }, source: "demo", syncing: false };
  }
  if (!backendConfigured) return { kind: "error", lastSync: null, retry: () => undefined, data: null };
  if (q.isError) return { kind: "error", lastSync: q.dataUpdatedAt || null, retry: () => void q.refetch(), data: q.data ?? null };
  if (q.data) return { kind: "ready", data: q.data, source: "live", syncing: !q.data.sync.complete };
  return { kind: "loading" };
}

/**
 * "@handle" for an idHash: remembered in this browser when the user tipped it, else from feed() events, else a short
 * hash (the chain stores X ids, not handles).
 */
export function useHandleNames() {
  const feed = useFeed("All", 25);
  const events = feed.kind === "ready" ? feed.data.events : feed.kind === "error" ? feed.data?.events ?? [] : [];
  const fromFeed = new Map(events.filter((e) => e.handle).map((e) => [e.idHash.toLowerCase(), e.handle as string]));
  return (idHash: string) => {
    const h = handleFor(idHash) ?? fromFeed.get(idHash.toLowerCase());
    return h ? `@${h}` : shortHash(idHash); // W4.1: the short idHash when the handle is unknown
  };
}

export type LookupState =
  | { kind: "idle" }
  | { kind: "invalid" }
  | { kind: "checking"; handle: string }
  | { kind: "found"; who: ResolvedHandle }
  | { kind: "missing"; handle: string }
  | { kind: "error"; handle: string; message: string };

/** Debounced display lookup for the send card (cached resolve). The tip flow re-resolves with `fresh` before signing. */
export function useHandleLookup(input: string, delayMs = 400): LookupState {
  const handle = normalizeHandle(input);
  const [debounced, setDebounced] = useState(handle);
  useEffect(() => {
    const t = setTimeout(() => setDebounced(handle), delayMs);
    return () => clearTimeout(t);
  }, [handle, delayMs]);
  const valid = HANDLE_RE.test(debounced);
  const q = useQuery({
    queryKey: ["resolve", debounced.toLowerCase()],
    queryFn: ({ signal }) => resolveHandle(debounced, { signal }),
    enabled: valid,
    staleTime: 60_000,
    retry: false,
  });
  if (!handle) return { kind: "idle" };
  if (!HANDLE_RE.test(handle)) return { kind: "invalid" };
  if (debounced !== handle || q.isFetching || q.isPending) return { kind: "checking", handle };
  if (q.isError) return { kind: "error", handle, message: apiErrorMessage(q.error, { handle }) };
  return q.data ? { kind: "found", who: q.data } : { kind: "missing", handle };
}

/** The tip flow's lookup: always fresh from X, never the cache (docs/backend.md, "Why the tip flow must use fresh"). */
export function useFreshResolve(handle: string | null) {
  return useQuery({
    queryKey: ["resolve-fresh", handle?.toLowerCase() ?? null],
    queryFn: ({ signal }) => resolveHandle(handle as string, { fresh: true, signal }),
    enabled: Boolean(handle && HANDLE_RE.test(handle)),
    staleTime: 0,
    gcTime: 0,
    retry: false,
  });
}
```

```ts
/** A user-typed DOGE amount -> wei, or null if it isn't a plain positive decimal with at most 18 decimals. */
export function parseDoge(input: string): bigint | null {
  const s = input.trim().replace(/,/g, "");
```

```tsx
/** Submitted -> confirmed, driven only by the receipt. Used right after signing and when resuming from ?tx=. */
function Outcome({ hash, handle, amountWei, note, onRetry, onConfirmed, onAnother }: { hash: Hex; handle: string; amountWei: bigint | null; note: string; onRetry: (() => void) | null; onConfirmed: (c: Confirmed | null) => void; onAnother: () => void }) {
  const client = usePublicClient({ chainId: CHAIN_ID }) as PublicClient;
  const { refundDelaySeconds } = useEscrowConfig();
  const wallet = useWallet();
  const [state, setState] = useState<{ kind: "waiting" } | { kind: "done"; block: bigint; time: number; fee: bigint | null; refundAt: number | null } | { kind: "reverted"; error: TxError } | { kind: "lost" }>({ kind: "waiting" });

  useEffect(() => {
    let live = true;
    client
      .waitForTransactionReceipt({ hash, timeout: 10 * 60_000 })
      .then(async (r) => {
        if (!live) return;
        if (r.status !== "success") {
          const error = await revertReason(client, hash, r.blockNumber);
          if (live) setState({ kind: "reverted", error });
          return;
        }
        const tipId = tipIdOf(r);
        const [block, fee, refundableAt] = await Promise.all([
          client.getBlock({ blockNumber: r.blockNumber }),
          feePaid(client, hash),
          tipId === null ? null : client.readContract({ address: ESCROW_ADDRESS, abi: escrowAbi, functionName: "refundableAt", args: [tipId] }).catch(() => null),
        ]);
        if (!live) return;
        const time = Number(block.timestamp);
        const refundAt = refundableAt === null ? null : Number(refundableAt);
        setState({ kind: "done", block: r.blockNumber, time, fee, refundAt });
        onConfirmed({ hash, time, tipId, refundAt, from: r.from });
        wallet.refetchBalance();
      })
      .catch(() => live && setState({ kind: "lost" }));
    return () => {
      live = false;
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [hash, client]);

  if (state.kind === "reverted" || state.kind === "lost")
    return (
      <div key="failed" className={card}>
        <div className="flex h-[52px] w-[52px] items-center justify-center rounded-[14px] bg-error-bg"><CloseIcon size={24} strokeWidth={2.6} className="text-error" /></div>

          {state.kind === "reverted" ? "The transaction was mined but the contract reverted it. Only the network fee was spent." : "No receipt after 10 minutes. Check it on the explorer before trying again, so you don't tip twice."}
        </p>
        {state.kind === "reverted" && (
          <div role="alert" className="mt-[18px] rounded-ctl border border-error-line bg-error-wash p-3.5 text-[14px] leading-[1.5]">
            <div className="font-semibold text-error">{state.error.title}</div>
            <div className="mt-1 text-ink-3">{state.error.detail}</div>
          </div>
        )}
        <TxRow hash={hash} />
        {onRetry && <button type="button" onClick={onRetry} className={`${gold} mt-4`}>Back to review</button>}
      </div>
    );

  if (state.kind === "waiting")
    return (
      <div key="waiting" className={card} aria-live="polite">
        <div className="flex items-center gap-2.5"><Spinner size={18} className="text-gold-spin" /><span className="text-[17px] font-semibold">Waiting for DogeOS</span></div>
        <p className="m-0 mt-2.5 text-[14px] leading-[1.5] text-ink-3">Your transaction is on its way. Blocks come every ~3 seconds. You can close this page; the tip will still go through.</p>
        <TxSteps included={false} />
        <TxRow hash={hash} />
      </div>
    );

  const amount = amountWei === null ? "Your tip" : `${formatDoge(amountWei)} DOGE`;
  const refundAtSec = state.refundAt ?? (refundDelaySeconds === undefined ? null : state.time + refundDelaySeconds);
  const refundAt = refundAtSec === null ? null : new Date(refundAtSec * 1000);
  return (
    <div key="done" className={card} aria-live="polite">
      <div className="flex h-[52px] w-[52px] items-center justify-center rounded-[14px] bg-success-bg"><CheckIcon size={26} strokeWidth={2.6} className="text-success" /></div>
      <div className="mt-4 text-[24px] font-bold tracking-[-0.02em]">{amount} is waiting for @{handle}</div>
      <p className="m-0 mt-2 text-[14px] leading-[1.5] text-ink-3">Confirmed in block {state.block.toLocaleString("en-US")}. Let them know so they can claim it.</p>
      <div className={`mt-[18px] flex flex-col gap-2 rounded-ctl border border-line-soft p-3.5 ${mono}`}>
        <div className="flex justify-between"><span className="text-muted">Fee paid</span><span>{state.fee === null ? "unknown" : `${formatDoge(state.fee, 4)} DOGE`}</span></div>
        <div className="flex justify-between"><span className="text-muted">Refundable from</span><span>{refundAt ? formatClock(refundAt) : "…"}</span></div>
        <div className="flex justify-between"><span className="text-muted">Tx</span><a href={explorerTx(hash)} target="_blank" rel="noreferrer" className="mt-link text-gold-ink">{shortHash(hash)} ↗</a></div>
      </div>
      <a href={xIntent(shareText(handle, amountWei, note))} target="_blank" rel="noreferrer" className={`${bigBtn} mt-4 border-ink bg-ink text-white`}><XLogo />Tell them on X</a>
      <button type="button" onClick={onAnother} className="mt-btn mt-2 h-11 w-full cursor-pointer border-0 bg-transparent text-[14px] font-semibold text-ink">Send another tip</button>
    </div>
  );
}
```

### `src/lib/reveal.ts`

```ts
/**
 * Sets data-in on [data-reveal] elements once they scroll into view, which plays their rise (index.css). An attribute
 * rather than a class, because React rewrites className on re-render but leaves attributes it didn't set alone.
 * One observer for the whole page; a MutationObserver picks up elements React renders later (route changes, data).
 * Does nothing unless index.html enabled motion (html.mt-motion).
 */
export function installReveal() {
  if (!document.documentElement.classList.contains("mt-motion")) return;
  const io = new IntersectionObserver(
    (entries) => {
      for (const e of entries) {
        if (!e.isIntersecting) continue;
        e.target.setAttribute("data-in", "");
        io.unobserve(e.target);
      }
    },
    { rootMargin: "0px 0px -8% 0px" },
  );
  const scan = () => {
    document.querySelectorAll<HTMLElement>("[data-reveal]:not([data-reveal-seen])").forEach((el) => {
      el.setAttribute("data-reveal-seen", "");
      io.observe(el);
    });
  };
  scan();
  new MutationObserver(scan).observe(document.body, { childList: true, subtree: true });
}
```

```tsx
type ClaimStep =
  | { kind: "idle" }
  | { kind: "signing" }
  | { kind: "wallet" }
  | { kind: "submitted"; hash: Hex }
  | { kind: "claimed"; hash: Hex; amountWei: bigint; tips: number; fee: bigint | null }
  | { kind: "failed"; error: TxError; reauth?: boolean };

const SESSION_CODES = new Set(["invalid_token", "missing_token"]);

/**
 * W3.4: ask sign-claim for a permission, simulate, sign, wait for the receipt. The permission expires about 300 s after
 * the chain's latest block; if the claim fails with Expired (the user waited too long in the wallet), a new permission
 * is fetched and the claim retried once, automatically.
 */
function useClaim(accessToken: string, sum: ReturnType<typeof useClaimSummary>) {
  const client = usePublicClient({ chainId: CHAIN_ID }) as PublicClient;
  const wallet = useWallet();
  const { writeContractAsync } = useWriteContract();
  const [step, setStep] = useState<ClaimStep>({ kind: "idle" });
  const claim = async () => {
    if (!wallet.address) return;
    const amountWei = sum.readyWei ?? 0n;
    const count = sum.pending.length;
    for (let attempt = 0; attempt < 2; attempt++) {
      const retry = (e: TxError) => e.revert === "Expired" && attempt === 0;
      setStep({ kind: "signing" });
      try {
        const sig = await signClaim(accessToken, wallet.address);
        if (!sig.claimable) {
          setStep({ kind: "failed", error: { kind: "revert", revert: "NothingToClaim", title: "Nothing to claim yet", detail: "There's nothing to claim right now." } });
          return;
        }
        const args = [BigInt(sig.xId), sig.recipient as Hex, BigInt(sig.nonce), BigInt(sig.expiry), sig.signature] as const;
        await client.simulateContract({ address: ESCROW_ADDRESS, abi: escrowAbi, functionName: "claim", args, account: wallet.address });
        setStep({ kind: "wallet" });
        const hash = await writeContractAsync({ address: ESCROW_ADDRESS, abi: escrowAbi, functionName: "claim", args, chainId: CHAIN_ID });
        setStep({ kind: "submitted", hash });
        const r = await client.waitForTransactionReceipt({ hash, timeout: 10 * 60_000 });
        if (r.status !== "success") {
          const error = await revertReason(client, hash, r.blockNumber);
          if (retry(error)) continue;
          setStep({ kind: "failed", error });
          return;
        }
        const fee = await feePaid(client, hash);
        setStep({ kind: "claimed", hash, amountWei, tips: count, fee });
        await Promise.all([sum.refetch(), wallet.refetchBalance()]);
        return;
      } catch (e) {
        if (e instanceof ApiError) {
          const reauth = SESSION_CODES.has(e.code);
          setStep({ kind: "failed", reauth, error: { kind: "rpc", title: reauth ? "Your X sign-in expired" : "Claiming didn't start", detail: apiErrorMessage(e) } });
          return;
        }
        const error = describeTxError(e);
        if (retry(error)) continue;
        setStep({ kind: "failed", error });
        return;
      }
    }
  };
  return { step, claim };
}
```

```tsx
        <div className="overflow-hidden rounded-card border border-line-2 bg-white">{rows}</div>
      )}
      {sent.hasMore && (
        <div className="mt-4 flex justify-center">
          <button type="button" onClick={sent.loadMore} disabled={sent.loadingMore} className="mt-btn inline-flex h-10 cursor-pointer items-center gap-2 rounded-ctl border border-line-input bg-white px-4 text-[13px] font-semibold text-ink">
            {sent.loadingMore ? <><Spinner size={14} />Loading…</> : "Show more"}
          </button>
        </div>
      )}
      {receipt && (
        <ReceiptDialog onClose={() => setReceipt(null)}>
          <ReceiptTicket
            status={receipt.status === "Claimed" ? "claimed" : receipt.status === "Refunded" ? "refunded" : "waiting"}
            amount={receipt.amountWei}
            from={receipt.tipper}
            to={nameOf(receipt.idHash)}
            sentAt={receipt.createdAt}
            refundAt={receipt.refundableAt}
            txHash={txOfTip.get(receipt.id.toString()) ?? null}
            notch="#F4F3EE"
            link={claimLink()}
            shareText={`I tipped ${nameOf(receipt.idHash)} ${formatDoge(receipt.amountWei)} DOGE on MuchTip. Claim it whenever you're ready: ${claimLink()}`}
          />
        </ReceiptDialog>
      )}
    </>
  );
}

function TipRow({ t, phone, name, tx, now, feeUpTo, balance, onReceipt, onDone }: { t: OnchainTip; phone: boolean; name: string; tx?: Hex; now: number; feeUpTo?: bigint; balance?: bigint; onReceipt: () => void; onDone: () => void }) {
  const client = usePublicClient({ chainId: CHAIN_ID }) as PublicClient;
  const { writeContractAsync } = useWriteContract();
  const toast = useToast();
  const [s, setS] = useState<RowState>({ kind: "idle" });
  const left = t.refundableAt - now;
  const cantPayFee = feeUpTo !== undefined && balance !== undefined && balance < feeUpTo;

  const ctx = { when: formatClock(new Date(t.refundableAt * 1000)) };
  const refund = async () => {
    setS({ kind: "wallet" });
    try {
      // a fresh quote for this exact refund(tipId) right before sending (FEES.md 5.2); the balance must cover it
      const data = encodeFunctionData({ abi: escrowAbi, functionName: "refund", args: [t.id] });
      const q = await quoteFee(client, "refund", { data, value: 0n }, t.tipper);
      if (balance !== undefined && balance < q.upToWei) {
        setS({ kind: "failed", message: `You need about ${formatFeeRange(null, q.upToWei).replace("up to ", "")} DOGE for the network fee.` });
        return;
      }
      await client.simulateContract({ address: ESCROW_ADDRESS, abi: escrowAbi, functionName: "refund", args: [t.id], account: t.tipper });
      const hash = await writeContractAsync({ address: ESCROW_ADDRESS, abi: escrowAbi, functionName: "refund", args: [t.id], chainId: CHAIN_ID });
      setS({ kind: "submitted" });
      const r = await client.waitForTransactionReceipt({ hash, timeout: 10 * 60_000 });
      if (r.status !== "success") {
        setS({ kind: "failed", message: (await revertReason(client, hash, r.blockNumber, ctx)).detail });
        return;
      }
      setS({ kind: "done" });
      toast(`Refunded ${formatDoge(t.amountWei, 2)} DOGE`, "success");
      onDone();
    } catch (e) {
      const error = describeTxError(e, ctx);
      setS({ kind: "failed", message: error.kind === "rejected" ? error.title : error.detail });
    }
  };

  const pill: TipPill = s.kind === "done" ? "refunded" : tipPillOf(t, now);
  const busy = s.kind === "wallet" || s.kind === "submitted";
  const amount = `${formatDoge(t.amountWei, 2)} DOGE`;
  const refundBtn = (
    <button type="button" onClick={refund} disabled={busy || cantPayFee} className={`mt-btn inline-flex cursor-pointer items-center gap-2 rounded-ctl border border-ink bg-gold font-semibold text-ink disabled:cursor-not-allowed disabled:border-line-2 disabled:bg-surface-chip disabled:text-muted-2 ${phone ? "h-[38px] px-3.5 text-[13px]" : "h-10 px-4 text-[14px]"}`}>
      {s.kind === "wallet" ? <><Spinner size={14} />Confirm in wallet</> : s.kind === "submitted" ? <><Spinner size={14} />Waiting for a block</> : "Refund"}
    </button>
  );
  const notes = (
    <>
      {s.kind === "failed" && <p role="alert" className="m-0 mt-2 basis-full text-[13px] text-error">{s.message}</p>}
      {pill === "refundable" && cantPayFee && s.kind === "idle" && feeUpTo !== undefined && (
        <p className="m-0 mt-2 basis-full text-[13px] text-warning">You need up to {formatFeeRange(null, feeUpTo).replace("up to ", "")} DOGE for the refund fee. <a href={FAUCET_URL} target="_blank" rel="noreferrer" className="font-semibold text-warning">Get test DOGE ↗</a></p>
      )}
    </>
  );
  const pillEl = (w: string) => <span className={`inline-flex h-7 items-center justify-center rounded-[8px] px-2.5 text-[12px] font-medium ${w} ${TIP_PILL[pill].cls}`}>{TIP_PILL[pill].label}</span>;

  if (phone)
    return (
      <div className="rounded-[14px] border border-line-2 bg-white p-3.5">
        <div className="flex items-center justify-between gap-3"><strong className="truncate">{name}</strong><span className="font-mono text-[14px] font-semibold">{amount}</span></div>
        <div className="mt-2.5 flex items-center justify-between gap-2.5">
          {pillEl("!h-[26px] !rounded-badge !px-[9px]")}
          {pill === "waiting" && <span className="font-mono text-[12px] font-medium text-muted">refund in {formatCountdown(left)}</span>}
          {pill === "refundable" && refundBtn}
        </div>
        {notes}
      </div>
    );

  return (
    <div className="mt-row flex flex-wrap items-center gap-4 border-b border-line-faint px-5 py-[18px] last:border-b-0">
      <div className="min-w-0 flex-[1_1_220px]">
        <div className="truncate font-semibold">{name}</div>
        <div className="font-mono text-[12px] text-muted">
          {formatDayLabel(new Date(t.createdAt * 1000))}
          {tx && <> · <a className="mt-link text-muted" href={explorerTx(tx)} target="_blank" rel="noreferrer">{shortHash(tx)} ↗</a></>}
        </div>
      </div>
      <span className="min-w-[90px] font-mono text-[15px] font-semibold">{amount}</span>
      {pillEl("w-[150px]")}
      <div className="flex w-[150px] justify-end">

        {pill === "refundable" && refundBtn}
        {pill === "claimed" && <button type="button" onClick={onReceipt} className="mt-btn h-10 cursor-pointer rounded-ctl border border-line-input bg-white px-3.5 text-[13px] font-semibold text-ink">Receipt</button>}
      </div>
      {notes}

  );
}
```

```tsx
/** Route changes start at the top, or at #anchor when the URL has one. A sign-in error on any page goes to Claim. */
function ScrollManager() {
  const { pathname, hash } = useLocation();
  useEffect(() => {
    // (scroll methods return a Promise in current Chrome, so never return their result from an effect)
    if (!hash) {
      window.scrollTo(0, 0);
      return;
    }
    const t = setTimeout(() => void document.getElementById(hash.slice(1))?.scrollIntoView(), 50);
    return () => clearTimeout(t);
  }, [pathname, hash]);
  return (
    <AuthErrorRedirect>
      <Suspense fallback={<div className="min-h-screen bg-paper" />}>
        <Outlet />
      </Suspense>
    </AuthErrorRedirect>
  );
}

/** /@handle is a tip link: Send with the handle filled in. Any other single segment is a 404. */
function HandleOr404() {
  const { segment = "" } = useParams();
  const handle = segment.startsWith("@") ? segment.slice(1) : "";
  return HANDLE_RE.test(handle) ? <Navigate to={`/app?to=${encodeURIComponent(handle)}`} replace /> : <NotFound />;
}

/**
 * v4 URLs keep working, query and hash included (the X sign-in return carries its result there). v4's /tip?to=&amount=
 * opened the review step, so that lands on Send's review (?step=review) rather than the empty form.
 */
function Moved({ to, review = false }: { to: string; review?: boolean }) {
  const { search, hash } = useLocation();
  const q = new URLSearchParams(search);
  if (review && q.get("to") && q.get("amount") && !q.get("tx")) q.set("step", "review");
  const qs = q.toString();
  return <Navigate to={`${to}${qs ? `?${qs}` : ""}${hash === "#send" ? "" : hash}`} replace />;
}
```
